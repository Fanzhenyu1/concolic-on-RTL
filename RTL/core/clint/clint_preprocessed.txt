module clint (    input wire clk,    input wire rst,    input wire [7:0] int_flag_i,    input wire [31:0] inst_i,    input wire [31:0] inst_addr_i,    input wire jump_flag_i,    input wire [31:0] jump_addr_i,    input wire div_started_i,    input wire [2:0] hold_flag_i,    input wire [31:0] data_i,    input wire [31:0] csr_mtvec,    input wire [31:0] csr_mepc,    input wire [31:0] csr_mstatus,    input wire global_int_en_i,    output wire hold_flag_o,    output reg        we_o,    output reg [31:0] waddr_o,    output reg [31:0] raddr_o,    output reg [31:0] data_o,    output reg [31:0] int_addr_o,    output reg        int_assert_o);
 reg [3:0] int_state;
 reg [4:0] csr_state;
 reg [31:0] inst_addr;
 reg [31:0] cause;
 assign hold_flag_o = ((int_state != 4'b0001) | (csr_state != 5'b00001)) ? 1'b1 : 1'b0;
always @(*) begin   if (rst == 1'b0) begin   int_state = 4'b0001;   end else begin   if (inst_i == 32'h73 || inst_i == 32'h00100073) begin   if (div_started_i == 1'b0) begin   int_state = 4'b0010;   end else begin   int_state = 4'b0001;   end   end else if (int_flag_i != 8'h0 && global_int_en_i == 1'b1) begin   int_state = 4'b0100;   end else if (inst_i == 32'h30200073) begin   int_state = 4'b1000;   end else begin   int_state = 4'b0001;   end   end   end
always @(posedge clk) begin   if (rst == 1'b0) begin   cause <= 32'h0;   end else if(csr_state == 5'b00001 && int_state == 4'b0010) begin   case (inst_i)   32'h73: begin   cause <= 32'd11;   end   32'h00100073: begin   cause <= 32'd3;   end   default: begin   cause <= 32'd10;   end   endcase   end else if (int_state == 4'b0100)   cause <= 32'h80000004;   end
always @(posedge clk) begin   if (rst == 1'b0) begin   csr_state <= 5'b00001;   inst_addr <= 32'h0;   end else begin   case (csr_state)   5'b00001: begin   if (int_state == 4'b0010) begin   csr_state <= 5'b00100;   if (jump_flag_i == 1'b1) begin   inst_addr <= jump_addr_i - 4'h4;   end else begin   inst_addr <= inst_addr_i;   end   end else if (int_state == 4'b0100) begin   csr_state <= 5'b00100;   if (jump_flag_i == 1'b1) begin   inst_addr <= jump_addr_i;   end else if (div_started_i == 1'b1) begin   inst_addr <= inst_addr_i - 4'h4;   end else begin   inst_addr <= inst_addr_i;   end   end else if (int_state == 4'b1000) begin   csr_state <= 5'b01000;   end   end   5'b00100: begin   csr_state <= 5'b00010;   end   5'b00010: begin   csr_state <= 5'b10000;   end   5'b10000: begin   csr_state <= 5'b00001;   end   5'b01000: begin   csr_state <= 5'b00001;   end   default: csr_state <= 5'b00001;   endcase   end   end
always @(posedge clk) begin   if (rst == 1'b0) begin   we_o <= 1'b0;   waddr_o <= 32'h0;   data_o <= 32'h0;   end else begin   case (csr_state)   5'b00100: begin   we_o <= 1'b1;   waddr_o <= {20'h0, 12'h341};   data_o <= inst_addr;   end   5'b10000: begin   we_o <= 1'b1;   waddr_o <= {20'h0, 12'h342};   data_o <= cause;   end   5'b00010: begin   we_o <= 1'b1;   waddr_o <= {20'h0, 12'h300};   data_o <= {csr_mstatus[31:4], 1'b0, csr_mstatus[2:0]};   end   5'b01000: begin   we_o <= 1'b1;   waddr_o <= {20'h0, 12'h300};   data_o <= {csr_mstatus[31:4], csr_mstatus[7], csr_mstatus[2:0]};   end   default: begin   we_o <= 1'b0;   waddr_o <= 32'h0;   data_o <= 32'h0;   end   endcase   end   end
always @(posedge clk) begin   if (rst == 1'b0) begin   int_assert_o <= 1'b0;   int_addr_o <= 32'h0;   end else begin   case (csr_state)   5'b10000: begin   int_assert_o <= 1'b1;   int_addr_o <= csr_mtvec;   end   5'b01000: begin   int_assert_o <= 1'b1;   int_addr_o <= csr_mepc;   end   default: begin   int_assert_o <= 1'b0;   int_addr_o <= 32'h0;   end   endcase   end   end
endmodule
