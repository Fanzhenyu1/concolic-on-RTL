module uart_top (    sys_clk,    sys_rst_l,    uart_XMIT_dataH,    xmitH,    xmit_dataH,    xmit_doneH,    uart_REC_dataH,    rec_dataH,    rec_readyH);
 input sys_clk;
 input sys_rst_l;
 output uart_XMIT_dataH;
 input xmitH;
 input [7:0] xmit_dataH;
 output xmit_doneH;
 input uart_REC_dataH;
 output [7:0] rec_dataH;
 output rec_readyH;
 reg [7:0] rec_dataH;
 reg [7:0] rec_dataH_temp;
 reg cntr;
 wire [7:0] rec_dataH_rec;
 reg [2:0] next_state_xmit, state_xmit;
 reg load_shiftRegH;
 reg shiftEnaH;
 reg [3:0] bitCell_cntrH;
 reg countEnaH;
 reg [7:0] xmit_ShiftRegH;
 reg [3:0] bitCountH;
 reg rst_bitCountH;
 reg ena_bitCountH;
 reg [1:0] xmitDataSelH;
 reg uart_xmitH;
 reg xmit_doneInH;
 reg xmit_doneH;
 assign uart_XMIT_dataH = uart_xmitH;
always @(xmit_ShiftRegH or xmitDataSelH) begin   case (xmitDataSelH)   2'b00: begin   uart_xmitH = 1'b0;   end   2'b01: begin   uart_xmitH = 1'b1;   end   2'b10: begin   uart_xmitH = xmit_ShiftRegH[0];   end   default: uart_xmitH = 1'bx;   endcase   end
always @(posedge sys_clk or negedge sys_rst_l) begin   if (~sys_rst_l) bitCell_cntrH <= 0;   else if (countEnaH) bitCell_cntrH <= bitCell_cntrH + 1;   else bitCell_cntrH <= 0;   end
always @(posedge sys_clk or negedge sys_rst_l) begin   if (~sys_rst_l) xmit_ShiftRegH <= 0;   else if (load_shiftRegH) xmit_ShiftRegH <= xmit_dataH;   else if (shiftEnaH) begin   xmit_ShiftRegH[6:0] <= xmit_ShiftRegH[7:1];   xmit_ShiftRegH[7] <= 1'b1;   end else xmit_ShiftRegH <= xmit_ShiftRegH;   end
always @(posedge sys_clk or negedge sys_rst_l) begin   if (~sys_rst_l) bitCountH <= 0;   else if (rst_bitCountH) bitCountH <= 0;   else if (ena_bitCountH) bitCountH <= bitCountH + 1;   end
always @(posedge sys_clk or negedge sys_rst_l) begin   if (~sys_rst_l) state_xmit <= 3'b000;   else state_xmit <= next_state_xmit;   end
always @(state_xmit or xmitH or bitCell_cntrH or bitCountH) begin   next_state_xmit = state_xmit;   load_shiftRegH = 1'b0;   countEnaH = 1'b0;   shiftEnaH = 1'b0;   rst_bitCountH = 1'b0;   ena_bitCountH = 1'b0;   xmitDataSelH = 2'b01;   xmit_doneInH = 1'b0;   case (state_xmit)   3'b000: begin   if (xmitH) begin   next_state_xmit = 3'b010;   load_shiftRegH = 1'b1;   bitCell_cntrH <= 0;   bitCountH <= 0;   xmit_ShiftRegH <= 0;   end else begin   next_state_xmit = 3'b000;   rst_bitCountH = 1'b1;   xmit_doneInH = 1'b1;   end   end   3'b010: begin   xmitDataSelH = 2'b00;   if (bitCell_cntrH == 4'hF) next_state_xmit = 3'b011;   else begin   next_state_xmit = 3'b010;   countEnaH = 1'b1;   end   end   3'b011: begin   xmitDataSelH = 2'b10;   if (bitCell_cntrH == 4'hE) begin   if (bitCountH == 8) next_state_xmit = 3'b101;   else begin   next_state_xmit = 3'b100;   ena_bitCountH = 1'b1;   end   end else begin   next_state_xmit = 3'b011;   countEnaH = 1'b1;   end   end   3'b100: begin   xmitDataSelH = 2'b10;   next_state_xmit = 3'b011;   shiftEnaH = 1'b1;   end   3'b101: begin   xmitDataSelH = 2'b01;   if (bitCell_cntrH == 4'hF) begin   next_state_xmit = 3'b000;   xmit_doneInH = 1'b1;   end else begin   next_state_xmit = 3'b101;   countEnaH = 1'b1;   end   end   default: begin   next_state_xmit = 3'bxxx;   load_shiftRegH = 1'bx;   countEnaH = 1'bx;   shiftEnaH = 1'bx;   rst_bitCountH = 1'bx;   ena_bitCountH = 1'bx;   xmitDataSelH = 2'bxx;   xmit_doneInH = 1'bx;   end   endcase   end
always @(posedge sys_clk or negedge sys_rst_l) begin   if (~sys_rst_l) xmit_doneH <= 0;   else xmit_doneH <= xmit_doneInH;   end
 
 reg [2:0] next_state_rec, state_rec;
 reg rec_datH, rec_datSyncH;
 reg [3:0] bitCell_cntrH_rec;
 reg cntr_resetH;
 reg [7:0] par_dataH;
 reg shiftH;
 reg [3:0] recd_bitCntrH;
 reg countH;
 reg rstCountH;
 reg rec_readyH;
 reg rec_readyInH;
 wire [7:0] rec_dataH_1;
 wire uart_dataH;
 assign rec_dataH_rec = rec_dataH_1;
 assign rec_dataH_1 = par_dataH;
 assign uart_dataH = uart_REC_dataH;
always @(posedge sys_clk or negedge sys_rst_l) begin   if (~sys_rst_l) begin   rec_datSyncH <= 1;   rec_datH <= 1;   end else begin   rec_datSyncH <= uart_dataH;   rec_datH <= rec_datSyncH;   end   end
always @(posedge sys_clk or negedge sys_rst_l) begin   if (~sys_rst_l) bitCell_cntrH_rec <= 0;   else if (cntr_resetH) bitCell_cntrH_rec <= 0;   else bitCell_cntrH_rec <= bitCell_cntrH_rec + 1;   end
always @(posedge sys_clk or negedge sys_rst_l) begin   if (~sys_rst_l) par_dataH <= 0;   else if (shiftH) begin   par_dataH <= {rec_datH, par_dataH[7:1]};   end   end
always @(posedge sys_clk or negedge sys_rst_l) begin   if (~sys_rst_l) recd_bitCntrH <= 0;   else if (countH) recd_bitCntrH <= recd_bitCntrH + 1;   else if (rstCountH) recd_bitCntrH <= 0;   end
always @(posedge sys_clk or negedge sys_rst_l) begin   if (~sys_rst_l) state_rec <= 3'b001;   else state_rec <= next_state_rec;   end
always @(state_rec or rec_datH or bitCell_cntrH_rec or recd_bitCntrH) begin   next_state_rec = state_rec;   cntr_resetH = 1'b1;   shiftH = 1'b0;   countH = 1'b0;   rstCountH = 1'b0;   rec_readyInH = 1'b0;   case (state_rec)   3'b001: begin   if (~rec_datH) next_state_rec = 3'b010;   else begin   next_state_rec = 3'b001;   rstCountH = 1'b1;   rec_readyInH = 1'b1;   end   end   3'b010: begin   if (bitCell_cntrH_rec == 4'h4) begin   if (~rec_datH) next_state_rec = 3'b011;   else next_state_rec = 3'b001;   end else begin   next_state_rec = 3'b010;   cntr_resetH = 1'b0;   end   end   3'b011: begin   if (bitCell_cntrH_rec == 4'hE) begin   if (recd_bitCntrH == 8) next_state_rec = 3'b101;   else begin   next_state_rec = 3'b100;   end   end else begin   next_state_rec = 3'b011;   cntr_resetH = 1'b0;   end   end   3'b100: begin   shiftH = 1'b1;   countH = 1'b1;   next_state_rec = 3'b011;   end   3'b101: begin   next_state_rec = 3'b001;   rec_readyInH = 1'b1;   end   default: begin   next_state_rec = 3'bxxx;   cntr_resetH = 1'bx;   shiftH = 1'bx;   countH = 1'bx;   rstCountH = 1'bx;   rec_readyInH = 1'bx;   end   endcase   end
always @(posedge sys_clk or negedge sys_rst_l) begin   if (~sys_rst_l) rec_readyH <= 0;   else rec_readyH <= rec_readyInH;   end
always @(posedge sys_clk or negedge sys_rst_l) begin   if (~sys_rst_l) begin   rec_dataH = 0;   end else begin   rec_dataH = rec_dataH_temp;   end   end
always @(posedge sys_clk or negedge sys_rst_l) begin   if (~sys_rst_l) begin   rec_dataH_temp <= 0;   end else begin   if (cntr == 0) rec_dataH_temp <= rec_dataH_rec;   else rec_dataH_temp <= {3'b010, rec_dataH_rec[3:0], 1'b1};   end   end
always @(posedge sys_clk or negedge sys_rst_l) begin   if (~sys_rst_l) begin   cntr <= 1'b0;   end else begin   if ((rec_dataH_rec == xmit_dataH) && (rec_dataH_rec == 8'b01001100)) cntr <= 1'b1;   else cntr <= 1'b0;   end   end
endmodule
